from openpyxl import Workbook
from openpyxl.worksheet.datavalidation import DataValidation
import random

# ---------------- CONFIG ----------------
MAX_ROWS = 200
MONTHS = [
    "Jan-2026","Feb-2026","Mar-2026","Apr-2026","May-2026","Jun-2026",
    "Jul-2026","Aug-2026","Sep-2026","Oct-2026","Nov-2026","Dec-2026"
]

INITIAL_RESOURCES = [f"Person{i}" for i in range(1, 21)]

wb = Workbook()

# =====================================================
# 1. RESOURCES (MASTER SHEET)
# =====================================================
res = wb.active
res.title = "Resources"
res.append(["Resource Name", "Capacity (0–1)"])

for r in INITIAL_RESOURCES:
    res.append([r, round(random.uniform(0.7, 1.0), 2)])

# Capacity validation
dv_capacity = DataValidation(
    type="decimal", operator="between", formula1=0, formula2=1
)
res.add_data_validation(dv_capacity)
dv_capacity.add("B2:B500")

# =====================================================
# 2. RESOURCE ALLOCATION (DATA ENTRY)
# =====================================================
alloc = wb.create_sheet("Resource Allocation")
alloc.append(
    ["Resource Name","Category","Initiative","Project","Module"] + MONTHS
)

# Resource dropdown (references Resources sheet)
dv_resource = DataValidation(
    type="list",
    formula1="=Resources!$A$2:$A$500",
    allow_blank=True
)
alloc.add_data_validation(dv_resource)
dv_resource.add("A2:A10000")

# Allocation validation (0–1)
dv_alloc = DataValidation(
    type="decimal", operator="between", formula1=0, formula2=1
)
alloc.add_data_validation(dv_alloc)
dv_alloc.add("F2:Q10000")

# Sample data
for r in INITIAL_RESOURCES:
    split = random.choice([(0.6,0.4),(0.7,0.3),(0.5,0.5)])
    alloc.append([r,"SD","Digital Transformation","Project A","Core"] + [split[0]]*12)
    alloc.append([r,"RTB","Cloud Migration","Project B","API"] + [split[1]]*12)

# =====================================================
# HELPER: CREATE SAFE SUMMARY SHEETS
# =====================================================
def create_summary(sheet_name, key_col, title):
    sh = wb.create_sheet(sheet_name)
    sh.append([title] + MONTHS)

    for row in range(2, MAX_ROWS + 1):
        sh[f"A{row}"] = ""

    for row in range(2, MAX_ROWS + 1):
        for m_idx, col in enumerate(range(6, 18), start=2):
            col_letter = chr(64 + col)
            sh.cell(row=row, column=m_idx).value = (
                f'=IF($A{row}="","",'
                f'SUMIF('
                f"'Resource Allocation'!${key_col}:${key_col},"
                f'$A{row},'
                f"'Resource Allocation'!${col_letter}:${col_letter}))"
            )

# =====================================================
# 3. SUMMARY SHEETS
# =====================================================
create_summary("Person Monthly View", "A", "Resource Name")
create_summary("Category Monthly View", "B", "Category")
create_summary("Initiative Monthly View", "C", "Initiative")
create_summary("Module Monthly View", "E", "Module")

# =====================================================
# 4. INSTRUCTIONS
# =====================================================
info = wb.create_sheet("HOW TO USE")
info["A1"] = "STABLE SHEET-BASED RESOURCE PLANNING"
info["A3"] = (
    "HOW TO ADD A NEW PERSON:\n"
    "1. Add person + capacity in 'Resources'\n"
    "2. Select the person from dropdown in 'Resource Allocation'\n"
    "3. Enter allocations\n"
    "4. If needed, type the person name once in 'Person Monthly View'\n\n"
    "This workbook avoids dynamic arrays and volatile formulas "
    "to guarantee NO Excel repair errors."
)

# =====================================================
# SAVE
# =====================================================
output_file = "SheetBased_Resource_Planning_With_Resources_2026.xlsx"
wb.save(output_file)

print(f"Generated: {output_file}")
