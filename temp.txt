# Excel 365–only dynamic array solution (UNIQUE + SUMIFS)
# Verified logic for Excel 365 where dynamic arrays are supported and stable.

from openpyxl import Workbook
from openpyxl.worksheet.datavalidation import DataValidation

wb = Workbook()

months = [
    "Jan-2026","Feb-2026","Mar-2026","Apr-2026","May-2026","Jun-2026",
    "Jul-2026","Aug-2026","Sep-2026","Oct-2026","Nov-2026","Dec-2026"
]

# ---------------- 1. Resources (Data Entry) ----------------
res = wb.active
res.title = "Resources"
res.append(["Name","Type","Capacity (0-1)"])

# sample data
res.append(["Person1","SCC",1])
res.append(["Person2","TW",0.8])

dv_type = DataValidation(type="list", formula1='"SCC,TW,Other"')
res.add_data_validation(dv_type)
dv_type.add("B2:B500")

dv_cap = DataValidation(type="decimal", operator="between", formula1=0, formula2=1)
res.add_data_validation(dv_cap)
dv_cap.add("C2:C500")

# ---------------- 2. Allocation (Data Entry) ----------------
alloc = wb.create_sheet("Allocation")
alloc.append(["Resource Name","Category","Initiative","Project","Module"] + months)

# Resource dropdown from Resources
dv_res = DataValidation(type="list", formula1="=Resources!$A$2:$A$500")
alloc.add_data_validation(dv_res)
dv_res.add("A2:A5000")

# Allocation validation
dv_alloc = DataValidation(type="decimal", operator="between", formula1=0, formula2=1)
alloc.add_data_validation(dv_alloc)
alloc.add_data_validation(dv_alloc)
dv_alloc.add("F2:Q5000")

# sample allocations
alloc.append(["Person1","SD","Init1","Proj1","Mod1"] + [0.5]*12)
alloc.append(["Person1","RTB","Init2","Proj2","Mod2"] + [0.5]*12)
alloc.append(["Person2","SD","Init1","Proj1","Mod1"] + [0.8]*12)

# -------- helper to build dynamic monthly views --------
def build_view(sheet_name, key_col, title):
    sh = wb.create_sheet(sheet_name)
    sh.append([title] + months)

    # dynamic key list
    sh["A2"] = f"=UNIQUE(Allocation!{key_col}2:{key_col}5000)"

    # monthly aggregations
    for i in range(len(months)):
        alloc_col = chr(ord("F") + i)
        sh_col = chr(ord("B") + i)
        sh[f"{sh_col}2"] = (
            f"=SUMIFS(Allocation!{alloc_col}2:{alloc_col}5000,"
            f"Allocation!{key_col}2:{key_col}5000,A2#)"
        )

# ---------------- 3. Views ----------------
build_view("Person Monthly View","A","Resource Name")
build_view("Category Monthly View","B","Category")
build_view("Initiative Monthly View","C","Initiative")
build_view("Module Monthly View","E","Module")

# ---------------- Save ----------------
path = "1MO_Resource_Planning_2026.xlsx"
wb.save(path)
